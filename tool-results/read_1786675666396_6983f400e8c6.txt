    41→/** Convert ASCII digits in a string/number to Persian digits. */

    52→/* ── useInViewOnce — one-shot scroll reveal ── */
    53→export function useInViewOnce<T extends HTMLElement = HTMLDivElement>(
    54→  options?: IntersectionObserverInit
    55→) {
    56→  const ref = useRef<T>(null);
    57→  const [inView, setInView] = useState(false);
    58→  useEffect(() => {
    59→    const el = ref.current;
    60→    if (!el || inView) return;
    61→    const obs = new IntersectionObserver(
    62→      (entries) => {
    63→        if (entries[0]?.isIntersecting) {
    64→          setInView(true);
    65→          obs.disconnect();
    66→        }
    67→      },
    68→      options ?? { threshold: 0.15, rootMargin: '0px 0px -40px 0px' }
    69→    );
    70→    obs.observe(el);
    71→    return () => obs.disconnect();
    72→  }, [inView, options]);
    73→  return { ref, inView };
    74→}

   102→/* ── useHaptic — light vibration on supported devices ──
   103→   Returns a `tap(ms?)` function. No-op where `navigator.vibrate` is
   104→   missing (desktop / iOS Safari) or when the user prefers reduced
   105→   motion (accessibility). Always safe to call. */
   106→export function useHaptic() {
   107→  return useCallback((ms: number | number[] = 8) => {
   108→    if (typeof window === 'undefined' || typeof navigator === 'undefined') return;
   109→    // Respect the user's reduced-motion preference (also covers vibration).
   110→    const mq = window.matchMedia?.('(prefers-reduced-motion: reduce)');
   111→    if (mq?.matches) return;
   112→    if (typeof navigator.vibrate === 'function') {
   113→      try { navigator.vibrate(ms); } catch { /* ignore */ }
   114→    }
   115→  }, []);
   116→}

   118→/* ── usePrefersReducedMotion — boolean hook ── */
   119→export function usePrefersReducedMotion() {
   120→  const [reduce, setReduce] = useState(false);
   121→  useEffect(() => {
   122→    if (typeof window === 'undefined' || !window.matchMedia) return;
   123→    const mq = window.matchMedia('(prefers-reduced-motion: reduce)');
   124→    // eslint-disable-next-line react-hooks/set-state-in-effect
   125→    setReduce(mq.matches);
   126→    const onChange = () => setReduce(mq.matches);
   127→    mq.addEventListener?.('change', onChange);
   128→    return () => mq.removeEventListener?.('change', onChange);
   129→  }, []);
   130→  return reduce;
   131→}

   134→   Pure CSS animation (no JS loop). Renders N absolutely-positioned

   136→   Disabled (renders nothing) when the user prefers reduced motion. */
   137→export function AmbientParticles({ count = 6 }: { count?: number }) {
   138→  const reduce = usePrefersReducedMotion();
   139→  // Trim the count on low-end devices for performance.
   140→  const n = typeof navigator !== 'undefined' && (navigator as Navigator & { hardwareConcurrency?: number }).hardwareConcurrency
   141→    ? Math.min(count, (navigator as Navigator & { hardwareConcurrency?: number }).hardwareConcurrency! < 4 ? 3 : count)
   142→    : count;

   144→  if (reduce) return null;

   146→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   147→      {dots.map((_, i) => {
   148→        const left = (i * 17 + 11) % 100;
   149→        const top = (i * 29 + 7) % 100;
   150→        const size = 4 + ((i * 7) % 6);
   151→        const dur = 9 + (i % 5) * 2;

   157→            initial={{ opacity: 0 }}
   158→            animate={{ opacity: [0, 0.5, 0], y: [0, -22, 0] }}

   162→              left: `${left}%`,

   176→/* ── useConfetti — a tiny one-shot confetti fired from a point ──
   177→   Call `fire()` from a parent (e.g. on share-copy). Renders a
   178→   fixed overlay; each piece is a small rotated div animated with
   179→   framer-motion, removed after ~1s. No canvas, no deps. Disabled
   180→   (fire is a no-op, node is null) when the user prefers reduced
   181→   motion. */
   182→export function useConfetti() {
   183→  const reduce = usePrefersReducedMotion();
   184→  const [pieces, setPieces] = useState<
   185→    { id: number; x: number; y: number; dx: number; dy: number; rot: number; color: string }[]
   186→  >([]);
   187→  const idRef = useRef(0);
   188→  const fire = useCallback((originX?: number, originY?: number) => {
   189→    if (reduce) return;
   190→    const colors = ['#F26A21', '#FBBF24', '#F59E0B', '#34D399', '#F472B6'];
   191→    const ox = originX ?? (typeof window !== 'undefined' ? window.innerWidth / 2 : 200);
   192→    const oy = originY ?? (typeof window !== 'undefined' ? window.innerHeight - 120 : 400);
   193→    const next = Array.from({ length: 18 }).map(() => {
   194→      const angle = (Math.random() * Math.PI) - Math.PI / 2; // upward fan
   195→      const dist = 60 + Math.random() * 120;
   196→      return {
   197→        id: idRef.current++,
   198→        x: ox,
   199→        y: oy,
   200→        dx: Math.cos(angle) * dist,
   201→        dy: Math.sin(angle) * dist - 40,
   202→        rot: (Math.random() - 0.5) * 540,
   203→        color: colors[Math.floor(Math.random() * colors.length)],
   204→      };
   205→    });
   206→    setPieces(next);
   207→    // Clear after the animation finishes (bounds duration to ~1.1s).
   208→    setTimeout(() => setPieces([]), 1100);
   209→  }, [reduce]);
   210→  const node =
   211→    pieces.length === 0 ? null : (
   212→      <div aria-hidden className="md:hidden fixed inset-0 z-[60] pointer-events-none">
   213→        {pieces.map((p) => (
   214→          <motion.span
   215→            key={p.id}
   216→            initial={{ x: p.x, y: p.y, opacity: 1, rotate: 0, scale: 1 }}
   217→            animate={{ x: p.x + p.dx, y: p.y + p.dy + 120, opacity: 0, rotate: p.rot, scale: 0.6 }}
   218→            transition={{ duration: 1, ease: [0.2, 0.6, 0.3, 1] }}
   219→            className="absolute w-2 h-2.5 rounded-[2px]"
   220→            style={{ background: p.color }}
   221→          />
   222→        ))}
   223→      </div>
   224→    );
   225→  return { fire, node };
   226→}

   228→/* ── useScrollRestore — persist scrollY to sessionStorage ──
   229→   Saves the page scroll position (throttled to every 400ms) under
   230→   the given key, and restores it on mount — UNLESS a location hash
   231→   is present (then the browser's native anchor jump wins). Returns
   232→   nothing; just call it once near the top of a client component. */
   233→export function useScrollRestore(key: string) {
   234→  useEffect(() => {
   235→    // Restore on mount, but defer to a deep-link hash if present (the
   236→    // browser's native anchor jump should win in that case).
   237→    const hasHash = typeof window !== 'undefined' && !!window.location.hash;
   238→    if (!hasHash) {
   239→      try {
   240→        const saved = sessionStorage.getItem(key);
   241→        if (saved) {
   242→          const y = Number(saved);
   243→          if (Number.isFinite(y) && y > 0) {
   244→            // Defer until after paint so the page has height.
   245→            requestAnimationFrame(() => window.scrollTo({ top: y, behavior: 'auto' }));
   246→          }
   247→        }
   248→      } catch { /* sessionStorage may be unavailable */ }
   249→    }
   250→
   251→    let t: ReturnType<typeof setTimeout> | null = null;
   252→    const onScroll = () => {
   253→      if (t) return;
   254→      t = setTimeout(() => {
   255→        t = null;
   256→        try { sessionStorage.setItem(key, String(window.scrollY)); } catch { /* ignore */ }
   257→      }, 400);
   258→    };
   259→    window.addEventListener('scroll', onScroll, { passive: true });
   260→    return () => {
   261→      window.removeEventListener('scroll', onScroll);
   262→      if (t) clearTimeout(t);
   263→    };
   264→  }, [key]);
   265→}

   267→/* ── useLocalStorage — generic typed localStorage state ──
   268→   SSR-safe (returns the initial value on the server). Syncs across
   269→   tabs via the `storage` event AND within the same tab via a custom
   270→   window event (the native `storage` event only fires cross-tab).
   271→   v6.1: the side effects (localStorage.setItem + event dispatch) are
   272→   moved OUT of the setState updater — running them inside the updater
   273→   caused a "Cannot update component A while rendering component B"
   274→   warning because React invokes the updater during render and the
   275→   dispatched event synchronously triggered other instances' setState.
   276→   A ref now tracks the current value so `next` can be computed in the
   277→   callback body (event-handler context), not in the render-phase updater. */
   278→export function useLocalStorage<T>(key: string, initial: T): [T, (v: T | ((prev: T) => T)) => void] {
   279→  const [value, setValue] = useState<T>(initial);
   280→  // Keep a ref of the latest value so the setter can compute `next`
   281→  // outside the render-phase updater (side effects must not run in render).
   282→  const valueRef = useRef(value);
   283→  useEffect(() => {
   284→    valueRef.current = value;
   285→  }, [value]);
   286→  useEffect(() => {
   287→    try {
   288→      const raw = localStorage.getItem(key);
   289→      if (raw !== null) {
   290→        const parsed = JSON.parse(raw) as T;
   291→        valueRef.current = parsed;
   292→        // eslint-disable-next-line react-hooks/set-state-in-effect
   293→        setValue(parsed);
   294→      }
   295→    } catch { /* ignore */ }
   296→    // Cross-tab sync.
   297→    const onStorage = (e: StorageEvent) => {
   298→      if (e.key === key && e.newValue !== null) {
   299→        try {
   300→          const parsed = JSON.parse(e.newValue) as T;
   301→          valueRef.current = parsed;
   302→          setValue(parsed);
   303→        } catch { /* ignore */ }
   304→      }
   305→    };
   306→    // Same-tab sync: a custom event dispatched by the setter below.
   307→    const onSameTab = (e: Event) => {
   308→      const detail = (e as CustomEvent).detail as { key: string; value: T };
   309→      if (detail && detail.key === key) {
   310→        valueRef.current = detail.value;
   311→        setValue(detail.value);
   312→      }
   313→    };
   314→    window.addEventListener('storage', onStorage);
   315→    window.addEventListener('mobile:local-storage', onSameTab as EventListener);
   316→    return () => {
   317→      window.removeEventListener('storage', onStorage);
   318→      window.removeEventListener('mobile:local-storage', onSameTab as EventListener);
   319→    };
   320→  }, [key]);
   321→  const set = useCallback(
   322→    (v: T | ((prev: T) => T)) => {
   323→      // Compute `next` in the callback body (event-handler context), NOT
   324→      // inside the setValue updater, so side effects don't run during render.
   325→      const next = typeof v === 'function' ? (v as (p: T) => T)(valueRef.current) : v;
   326→      valueRef.current = next;
   327→      setValue(next);
   328→      try {
   329→        localStorage.setItem(key, JSON.stringify(next));
   330→        // Notify other hook instances in THIS tab.
   331→        window.dispatchEvent(new CustomEvent('mobile:local-storage', { detail: { key, value: next } }));
   332→      } catch { /* ignore */ }
   333→    },
   334→    [key]
   335→  );
   336→  return [value, set];
   337→}

   339→/* ── useSessionStorage — like useLocalStorage but for sessionStorage.
   340→   Used for ephemeral per-session state (e.g. recently-viewed sections)
   341→   that should survive a reload but not a new tab/session. Same-tab
   342→   sync via a custom event so multiple instances stay consistent. */
   343→export function useSessionStorage<T>(key: string, initial: T): [T, (v: T | ((prev: T) => T)) => void] {
   344→  const [value, setValue] = useState<T>(initial);
   345→  const valueRef = useRef(value);
   346→  useEffect(() => {
   347→    valueRef.current = value;
   348→  }, [value]);
   349→  useEffect(() => {
   350→    try {
   351→      const raw = sessionStorage.getItem(key);
   352→      if (raw !== null) {
   353→        const parsed = JSON.parse(raw) as T;
   354→        valueRef.current = parsed;
   355→        // eslint-disable-next-line react-hooks/set-state-in-effect
   356→        setValue(parsed);
   357→      }
   358→    } catch { /* ignore */ }
   359→    const onSameTab = (e: Event) => {
   360→      const detail = (e as CustomEvent).detail as { key: string; value: T };
   361→      if (detail && detail.key === key) {
   362→        valueRef.current = detail.value;
   363→        setValue(detail.value);
   364→      }
   365→    };
   366→    window.addEventListener('mobile:session-storage', onSameTab as EventListener);
   367→    return () => window.removeEventListener('mobile:session-storage', onSameTab as EventListener);
   368→  }, [key]);
   369→  const set = useCallback(
   370→    (v: T | ((prev: T) => T)) => {
   371→      const next = typeof v === 'function' ? (v as (p: T) => T)(valueRef.current) : v;
   372→      valueRef.current = next;
   373→      setValue(next);
   374→      try {
   375→        sessionStorage.setItem(key, JSON.stringify(next));
   376→        window.dispatchEvent(new CustomEvent('mobile:session-storage', { detail: { key, value: next } }));
   377→      } catch { /* ignore */ }
   378→    },
   379→    [key]
   380→  );
   381→  return [value, set];
   382→}

   384→/* ── useBookmarks — persist a Set of section ids to localStorage ──
   385→   Returns `{ bookmarks, toggle, has }`. `bookmarks` is a string[] for
   386→   easy rendering; `toggle(id)` adds/removes; `has(id)` checks membership. */
   387→export function useBookmarks(storageKey: string) {
   388→  const [arr, setArr] = useLocalStorage<string[]>(storageKey, []);
   389→  const set = useMemo(() => new Set(arr), [arr]);
   390→  const has = useCallback((id: string) => set.has(id), [set]);
   391→  const toggle = useCallback(
   392→    (id: string) => {
   393→      setArr((prev) => {
   394→        const s = new Set(prev);
   395→        if (s.has(id)) s.delete(id);
   396→        else s.add(id);
   397→        return Array.from(s);
   398→      });
   399→    },
   400→    [setArr]
   401→  );
   402→  return { bookmarks: arr, toggle, has };
   403→}

   405→/* ── useSectionRead — time-weighted section-read tracking.
   406→   Accumulates viewing time while a section is ≥30% visible; marks it
   407→   read after 3s of accumulated viewing (across any number of visits).
   408→   This works for BOTH short sections (which reach 30% easily and stay)
   409→   AND tall sections (which the user scrolls through, accumulating 30%+
   410→   visibility across multiple viewport-fuls). Persists to localStorage.
   411→   Returns `{ read: Set<string>, isRead }`. */
   412→export function useSectionRead(ids: string[], storageKey: string) {
   413→  const [arr, setArr] = useLocalStorage<string[]>(storageKey, []);
   414→  const readSet = useMemo(() => new Set(arr), [arr]);
   415→  const elapsedRef = useRef<Record<string, number>>({});
   416→  const visibleRef = useRef<Set<string>>(new Set());
   417→
   418→  useEffect(() => {
   419→    if (ids.length === 0) return;
   420→    const observers: IntersectionObserver[] = [];
   421→    ids.forEach((id) => {
   422→      const el = document.getElementById(id);
   423→      if (!el) return;
   424→      const obs = new IntersectionObserver(
   425→        (entries) => {
   426→          const e = entries[0];
   427→          if (!e) return;
   428→          if (e.intersectionRatio >= 0.3) visibleRef.current.add(id);
   429→          else visibleRef.current.delete(id);
   430→        },
   431→        { threshold: [0, 0.3, 0.6] }
   432→      );
   433→      obs.observe(el);
   434→      observers.push(obs);
   435→    });
   436→
   437→    // Accumulate 250ms per tick while visible; mark read at 3000ms.
   438→    const READ_THRESHOLD_MS = 3000;
   439→    const interval = setInterval(() => {
   440→      const visible = visibleRef.current;
   441→      if (visible.size === 0) return;
   442→      const toAdd: string[] = [];
   443→      visible.forEach((id) => {
   444→        elapsedRef.current[id] = (elapsedRef.current[id] || 0) + 250;
   445→        if (elapsedRef.current[id] >= READ_THRESHOLD_MS) {
   446→          toAdd.push(id);
   447→        }
   448→      });
   449→      if (toAdd.length > 0) {
   450→        setArr((prev) => {
   451→          const existing = new Set(prev);
   452→          const next = [...prev];
   453→          toAdd.forEach((id) => {
   454→            if (!existing.has(id)) {
   455→              next.push(id);
   456→              delete elapsedRef.current[id];
   457→              visibleRef.current.delete(id);
   458→            }
   459→          });
   460→          return next.length === prev.length ? prev : next;
   461→        });
   462→      }
   463→    }, 250);
   464→
   465→    return () => {
   466→      observers.forEach((o) => o.disconnect());
   467→      clearInterval(interval);
   468→    };
   469→  }, [ids, setArr]);
   470→
   471→  const isRead = useCallback((id: string) => readSet.has(id), [readSet]);
   472→  return { read: readSet, isRead };
   473→}

   475→/* ── useReadTimestamps — tracks WHEN each section was read (epoch ms),
   476→   persisted to localStorage. Used by the reading-streak indicator
   477→   ("X بخش امروز"). Returns `{ timestamps, markRead, countToday, clear }`.
   478→   `markRead(id)` records the current time for a section (idempotent —
   479→   only records the FIRST read time). `countToday` returns how many
   480→   sections were first-read today. `clear` wipes the timestamps. */
   481→export function useReadTimestamps(storageKey: string) {
   482→  const [map, setMap] = useLocalStorage<Record<string, number>>(storageKey, {});
   483→  const markRead = useCallback(
   484→    (id: string) => {
   485→      setMap((prev) => (prev[id] ? prev : { ...prev, [id]: Date.now() }));
   486→    },
   487→    [setMap]
   488→  );
   489→  const clear = useCallback(() => setMap({}), [setMap]);
   490→  const countToday = useMemo(() => {
   491→    const now = new Date();
   492→    const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
   493→    return Object.values(map).filter((ts) => ts >= startOfToday).length;
   494→  }, [map]);
   495→  return { timestamps: map, markRead, countToday, clear };
   496→}

   498→/* ── ProgressRing — a small SVG ring showing a 0..1 progress.
   499→   Used on the bookmarks badge to show read-progress visually.
   500→   v8: when progress is between 0 and 1 (exclusive), the ring gently
   501→   "breathes" (scales 1 ↔ 1.06) to draw attention to progress. */
   502→export function ProgressRing({
   503→  progress,
   504→  size = 18,
   505→  stroke = 2.5,
   506→  className = '',
   507→}: {
   508→  progress: number; // 0..1
   509→  size?: number;
   510→  stroke?: number;
   511→  className?: string;
   512→}) {
   513→  const r = (size - stroke) / 2;
   514→  const c = 2 * Math.PI * r;
   515→  const clamped = Math.max(0, Math.min(1, progress));
   516→  const breathing = clamped > 0 && clamped < 1;
   517→  return (

  1184→/* ── PullToRefreshHint — a subtle spinner that appears when the user
  1185→   overscrolls at the very top of the page. Mobile-only, pure visual
  1186→   (does NOT actually refresh — it's a delightful hint that the page
  1187→   is "springy"). Fades in when document is at scrollTop 0 AND a
  1188→   touch-move pulls downward; fades out otherwise. Uses a scroll
  1189→   listener on the body's overscroll as a lightweight proxy. */
  1190→export function PullToRefreshHint() {
  1191→  const [pull, setPull] = useState(0);
  1192→  useEffect(() => {
  1193→    let raf = 0;
  1194→    const onScroll = () => {
  1195→      cancelAnimationFrame(raf);
  1196→      raf = requestAnimationFrame(() => {
  1197→        // Only meaningful at the very top: a negative scrollY (rubber-band)
  1198→        // or a touch overscroll. We approximate with scrollY <= 0.
  1199→        const y = window.scrollY;
  1200→        setPull(y <= 0 ? Math.min(1, (Math.abs(y) + 1) / 60) : 0);
  1201→      });
  1202→    };
  1203→    window.addEventListener('scroll', onScroll, { passive: true });
  1204→    return () => {
  1205→      window.removeEventListener('scroll', onScroll);
  1206→      cancelAnimationFrame(raf);
  1207→    };
  1208→  }, []);
  1209→  if (pull <= 0.02) return null;
  1210→  return (
  1211→    <div
  1212→      aria-hidden
  1213→      className="md:hidden fixed top-1 left-1/2 -translate-x-1/2 z-30 pointer-events-none"
  1214→      style={{ opacity: pull }}
  1215→    >
  1216→      <motion.span
  1217→        animate={{ rotate: 360 }}
  1218→        transition={{ duration: 0.9, repeat: Infinity, ease: 'linear' }}

  1225→/* ── useSnapActive — tracks which SnapRail card is centered ──
  1226→   Attach the returned ref to a SnapRail's inner scroll container.
  1227→   Returns the active index. */
  1228→export function useSnapActive(count: number) {
  1229→  const ref = useRef<HTMLDivElement>(null);
  1230→  const [active, setActive] = useState(0);
  1231→  const onScroll = useCallback(() => {
  1232→    const el = ref.current;
  1233→    if (!el) return;
  1234→    const center = el.scrollLeft + el.clientWidth / 2;
  1235→    let best = 0;
  1236→    let bestDist = Infinity;
  1237→    Array.from(el.children).forEach((child, i) => {
  1238→      const c = child as HTMLElement;
  1239→      const cCenter = c.offsetLeft + c.clientWidth / 2;
  1240→      const d = Math.abs(cCenter - center);
  1241→      if (d < bestDist) {
  1242→        bestDist = d;
  1243→        best = i;
  1244→      }
  1245→    });
  1246→    setActive(best);
  1247→  }, [count]);
  1248→  useEffect(() => {
  1249→    const el = ref.current;
  1250→    if (!el) return;
  1251→    el.addEventListener('scroll', onScroll, { passive: true });
  1252→    // Defer the initial active-card computation to the next frame so we
  1253→    // don't call setState synchronously inside this effect (cascading render).
  1254→    const raf = requestAnimationFrame(onScroll);
  1255→    return () => {
  1256→      el.removeEventListener('scroll', onScroll);
  1257→      cancelAnimationFrame(raf);
  1258→    };
  1259→  }, [onScroll]);
  1260→  const scrollTo = useCallback(
  1261→    (i: number) => {
  1262→      const el = ref.current;
  1263→      if (!el) return;
  1264→      const child = el.children[i] as HTMLElement | undefined;
  1265→      if (!child) return;
  1266→      el.scrollTo({ left: child.offsetLeft - (el.clientWidth - child.clientWidth) / 2, behavior: 'smooth' });
  1267→    },
  1268→    []
  1269→  );
  1270→  return { ref, active, scrollTo };
  1271→}
