    36→/** Convert ASCII digits in a string/number to Persian digits. */
    37→export function toFa(input: string | number): string {
    38→  const fa = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    39→  return String(input).replace(/[0-9]/g, (d) => fa[Number(d)]);

    47→/* ── useInViewOnce — one-shot scroll reveal ── */
    48→export function useInViewOnce<T extends HTMLElement = HTMLDivElement>(
    49→  options?: IntersectionObserverInit
    50→) {
    51→  const ref = useRef<T>(null);
    52→  const [inView, setInView] = useState(false);
    53→  useEffect(() => {
    54→    const el = ref.current;
    55→    if (!el || inView) return;
    56→    const obs = new IntersectionObserver(
    57→      (entries) => {
    58→        if (entries[0]?.isIntersecting) {
    59→          setInView(true);
    60→          obs.disconnect();
    61→        }
    62→      },
    63→      options ?? { threshold: 0.15, rootMargin: '0px 0px -40px 0px' }
    64→    );
    65→    obs.observe(el);
    66→    return () => obs.disconnect();
    67→  }, [inView, options]);
    68→  return { ref, inView };
    69→}

    97→/* ── useHaptic — light vibration on supported devices ──
    98→   Returns a `tap(ms?)` function. No-op where `navigator.vibrate` is
    99→   missing (desktop / iOS Safari) or when the user prefers reduced
   100→   motion (accessibility). Always safe to call. */
   101→export function useHaptic() {
   102→  return useCallback((ms: number | number[] = 8) => {
   103→    if (typeof window === 'undefined' || typeof navigator === 'undefined') return;
   104→    // Respect the user's reduced-motion preference (also covers vibration).
   105→    const mq = window.matchMedia?.('(prefers-reduced-motion: reduce)');
   106→    if (mq?.matches) return;
   107→    if (typeof navigator.vibrate === 'function') {
   108→      try { navigator.vibrate(ms); } catch { /* ignore */ }
   109→    }
   110→  }, []);
   111→}

   113→/* ── usePrefersReducedMotion — boolean hook ── */
   114→export function usePrefersReducedMotion() {
   115→  const [reduce, setReduce] = useState(false);
   116→  useEffect(() => {
   117→    if (typeof window === 'undefined' || !window.matchMedia) return;
   118→    const mq = window.matchMedia('(prefers-reduced-motion: reduce)');
   119→    // eslint-disable-next-line react-hooks/set-state-in-effect
   120→    setReduce(mq.matches);
   121→    const onChange = () => setReduce(mq.matches);
   122→    mq.addEventListener?.('change', onChange);
   123→    return () => mq.removeEventListener?.('change', onChange);
   124→  }, []);
   125→  return reduce;
   126→}

   128→/* ── AmbientParticles — slow-floating dots for dark sections ──
   129→   Pure CSS animation (no JS loop). Renders N absolutely-positioned
   130→   dots with staggered delays. Pointer-events-none, aria-hidden.
   131→   Disabled (renders nothing) when the user prefers reduced motion. */
   132→export function AmbientParticles({ count = 6 }: { count?: number }) {
   133→  const reduce = usePrefersReducedMotion();
   134→  // Trim the count on low-end devices for performance.
   135→  const n = typeof navigator !== 'undefined' && (navigator as Navigator & { hardwareConcurrency?: number }).hardwareConcurrency
   136→    ? Math.min(count, (navigator as Navigator & { hardwareConcurrency?: number }).hardwareConcurrency! < 4 ? 3 : count)
   137→    : count;
   138→  const dots = Array.from({ length: reduce ? 0 : n });
   139→  if (reduce) return null;
   140→  return (
   141→    <div aria-hidden className="pointer-events-none absolute inset-0 overflow-hidden">
   142→      {dots.map((_, i) => {
   143→        const left = (i * 17 + 11) % 100;
   144→        const top = (i * 29 + 7) % 100;
   145→        const size = 4 + ((i * 7) % 6);
   146→        const dur = 9 + (i % 5) * 2;
   147→        const delay = (i % 4) * 1.3;
   148→        const orange = i % 2 === 0;
   149→        return (
   150→          <motion.span
   151→            key={i}
   152→            initial={{ opacity: 0 }}
   153→            animate={{ opacity: [0, 0.5, 0], y: [0, -22, 0] }}
   154→            transition={{ duration: dur, delay, repeat: Infinity, ease: 'easeInOut' }}
   155→            className="absolute rounded-full"
   156→            style={{
   157→              left: `${left}%`,
   158→              top: `${top}%`,
   159→              width: size,
   160→              height: size,
   161→              background: orange ? '#F26A21' : '#93C5FD',
   162→              filter: 'blur(1px)',
   163→            }}
   164→          />
   165→        );
   166→      })}
   167→    </div>

   171→/* ── useConfetti — a tiny one-shot confetti fired from a point ──
   172→   Call `fire()` from a parent (e.g. on share-copy). Renders a
   173→   fixed overlay; each piece is a small rotated div animated with
   174→   framer-motion, removed after ~1s. No canvas, no deps. Disabled
   175→   (fire is a no-op, node is null) when the user prefers reduced
   176→   motion. */
   177→export function useConfetti() {
   178→  const reduce = usePrefersReducedMotion();
   179→  const [pieces, setPieces] = useState<
   180→    { id: number; x: number; y: number; dx: number; dy: number; rot: number; color: string }[]
   181→  >([]);
   182→  const idRef = useRef(0);
   183→  const fire = useCallback((originX?: number, originY?: number) => {
   184→    if (reduce) return;
   185→    const colors = ['#F26A21', '#FBBF24', '#60A5FA', '#34D399', '#F472B6'];
   186→    const ox = originX ?? (typeof window !== 'undefined' ? window.innerWidth / 2 : 200);
   187→    const oy = originY ?? (typeof window !== 'undefined' ? window.innerHeight - 120 : 400);
   188→    const next = Array.from({ length: 18 }).map(() => {
   189→      const angle = (Math.random() * Math.PI) - Math.PI / 2; // upward fan
   190→      const dist = 60 + Math.random() * 120;
   191→      return {
   192→        id: idRef.current++,
   193→        x: ox,
   194→        y: oy,
   195→        dx: Math.cos(angle) * dist,
   196→        dy: Math.sin(angle) * dist - 40,
   197→        rot: (Math.random() - 0.5) * 540,
   198→        color: colors[Math.floor(Math.random() * colors.length)],
   199→      };
   200→    });
   201→    setPieces(next);
   202→    // Clear after the animation finishes (bounds duration to ~1.1s).
   203→    setTimeout(() => setPieces([]), 1100);
   204→  }, [reduce]);
   205→  const node =
   206→    pieces.length === 0 ? null : (
   207→      <div aria-hidden className="md:hidden fixed inset-0 z-[60] pointer-events-none">
   208→        {pieces.map((p) => (
   209→          <motion.span
   210→            key={p.id}
   211→            initial={{ x: p.x, y: p.y, opacity: 1, rotate: 0, scale: 1 }}
   212→            animate={{ x: p.x + p.dx, y: p.y + p.dy + 120, opacity: 0, rotate: p.rot, scale: 0.6 }}
   213→            transition={{ duration: 1, ease: [0.2, 0.6, 0.3, 1] }}
   214→            className="absolute w-2 h-2.5 rounded-[2px]"
   215→            style={{ background: p.color }}
   216→          />
   217→        ))}
   218→      </div>
   219→    );
   220→  return { fire, node };
   221→}

   223→/* ── useScrollRestore — persist scrollY to sessionStorage ──
   224→   Saves the page scroll position (throttled to every 400ms) under
   225→   the given key, and restores it on mount — UNLESS a location hash
   226→   is present (then the browser's native anchor jump wins). Returns
   227→   nothing; just call it once near the top of a client component. */
   228→export function useScrollRestore(key: string) {
   229→  useEffect(() => {
   230→    // Restore on mount, but defer to a deep-link hash if present (the
   231→    // browser's native anchor jump should win in that case).
   232→    const hasHash = typeof window !== 'undefined' && !!window.location.hash;
   233→    if (!hasHash) {
   234→      try {
   235→        const saved = sessionStorage.getItem(key);
   236→        if (saved) {
   237→          const y = Number(saved);
   238→          if (Number.isFinite(y) && y > 0) {
   239→            // Defer until after paint so the page has height.
   240→            requestAnimationFrame(() => window.scrollTo({ top: y, behavior: 'auto' }));
   241→          }
   242→        }
   243→      } catch { /* sessionStorage may be unavailable */ }
   244→    }
   245→
   246→    let t: ReturnType<typeof setTimeout> | null = null;
   247→    const onScroll = () => {
   248→      if (t) return;
   249→      t = setTimeout(() => {
   250→        t = null;
   251→        try { sessionStorage.setItem(key, String(window.scrollY)); } catch { /* ignore */ }
   252→      }, 400);
   253→    };
   254→    window.addEventListener('scroll', onScroll, { passive: true });
   255→    return () => {
   256→      window.removeEventListener('scroll', onScroll);
   257→      if (t) clearTimeout(t);
   258→    };
   259→  }, [key]);
   260→}

   262→/* ── useLocalStorage — generic typed localStorage state ──
   263→   SSR-safe (returns the initial value on the server). Syncs across
   264→   tabs via the `storage` event AND within the same tab via a custom
   265→   window event (the native `storage` event only fires cross-tab). */
   266→export function useLocalStorage<T>(key: string, initial: T): [T, (v: T | ((prev: T) => T)) => void] {
   267→  const [value, setValue] = useState<T>(initial);
   268→  useEffect(() => {
   269→    try {
   270→      const raw = localStorage.getItem(key);
   271→      if (raw !== null) {
   272→        // eslint-disable-next-line react-hooks/set-state-in-effect
   273→        setValue(JSON.parse(raw) as T);
   274→      }
   275→    } catch { /* ignore */ }
   276→    // Cross-tab sync.
   277→    const onStorage = (e: StorageEvent) => {
   278→      if (e.key === key && e.newValue !== null) {
   279→        try { setValue(JSON.parse(e.newValue) as T); } catch { /* ignore */ }
   280→      }
   281→    };
   282→    // Same-tab sync: a custom event dispatched by the setter below.
   283→    const onSameTab = (e: Event) => {
   284→      const detail = (e as CustomEvent).detail as { key: string; value: T };
   285→      if (detail && detail.key === key) setValue(detail.value);
   286→    };
   287→    window.addEventListener('storage', onStorage);
   288→    window.addEventListener('mobile:local-storage', onSameTab as EventListener);
   289→    return () => {
   290→      window.removeEventListener('storage', onStorage);
   291→      window.removeEventListener('mobile:local-storage', onSameTab as EventListener);
   292→    };
   293→  }, [key]);
   294→  const set = useCallback(
   295→    (v: T | ((prev: T) => T)) => {
   296→      setValue((prev) => {
   297→        const next = typeof v === 'function' ? (v as (p: T) => T)(prev) : v;
   298→        try {
   299→          localStorage.setItem(key, JSON.stringify(next));
   300→          // Notify other hook instances in THIS tab.
   301→          window.dispatchEvent(new CustomEvent('mobile:local-storage', { detail: { key, value: next } }));
   302→        } catch { /* ignore */ }
   303→        return next;
   304→      });
   305→    },
   306→    [key]
   307→  );
   308→  return [value, set];
   309→}

   311→/* ── useBookmarks — persist a Set of section ids to localStorage ──
   312→   Returns `{ bookmarks, toggle, has }`. `bookmarks` is a string[] for
   313→   easy rendering; `toggle(id)` adds/removes; `has(id)` checks membership. */
   314→export function useBookmarks(storageKey: string) {
   315→  const [arr, setArr] = useLocalStorage<string[]>(storageKey, []);
   316→  const set = useMemo(() => new Set(arr), [arr]);
   317→  const has = useCallback((id: string) => set.has(id), [set]);
   318→  const toggle = useCallback(
   319→    (id: string) => {
   320→      setArr((prev) => {
   321→        const s = new Set(prev);
   322→        if (s.has(id)) s.delete(id);
   323→        else s.add(id);
   324→        return Array.from(s);
   325→      });
   326→    },
   327→    [setArr]
   328→  );
   329→  return { bookmarks: arr, toggle, has };
   330→}

   332→/* ── useSectionRead — tracks sections the visitor has substantially
   333→   viewed (≥60% in view for >1.5s) and persists them to localStorage.
   334→   Returns `{ read: Set<string>, isRead }`. Pass the list of section
   335→   ids to observe. (v5: threshold relaxed from 80%/2s → 60%/1.5s so
   336→   tall sections like TopicFinder can also be marked read.) */
   337→export function useSectionRead(ids: string[], storageKey: string) {
   338→  const [arr, setArr] = useLocalStorage<string[]>(storageKey, []);
   339→  const readSet = useMemo(() => new Set(arr), [arr]);
   340→  const timersRef = useRef<Record<string, ReturnType<typeof setTimeout>>>({});
   341→  useEffect(() => {
   342→    if (ids.length === 0) return;
   343→    const observers: IntersectionObserver[] = [];
   344→    ids.forEach((id) => {
   345→      const el = document.getElementById(id);
   346→      if (!el) return;
   347→      const obs = new IntersectionObserver(
   348→        (entries) => {
   349→          const e = entries[0];
   350→          if (!e) return;
   351→          if (e.intersectionRatio >= 0.6) {
   352→            // Start a 1.5s timer; only mark read if it stays in view.
   353→            if (!timersRef.current[id]) {
   354→              timersRef.current[id] = setTimeout(() => {
   355→                setArr((prev) => (prev.includes(id) ? prev : [...prev, id]));
   356→                delete timersRef.current[id];
   357→              }, 1500);
   358→            }
   359→          } else {
   360→            // Left the viewport before 1.5s — cancel.
   361→            if (timersRef.current[id]) {
   362→              clearTimeout(timersRef.current[id]);
   363→              delete timersRef.current[id];
   364→            }
   365→          }
   366→        },
   367→        { threshold: [0, 0.6] }
   368→      );
   369→      obs.observe(el);
   370→      observers.push(obs);
   371→    });
   372→    return () => {
   373→      observers.forEach((o) => o.disconnect());
   374→      Object.values(timersRef.current).forEach(clearTimeout);
   375→      timersRef.current = {};
   376→    };
   377→  }, [ids, setArr]);
   378→  const isRead = useCallback((id: string) => readSet.has(id), [readSet]);
   379→  return { read: readSet, isRead };
   380→}

   497→/* ── ShimmerDivider — a SectionDivider variant with a moving
   498→   shimmer line instead of (or beside) the icon medallion. Use
   499→   where a richer seam is wanted. */
   500→export function ShimmerDivider({
   501→  variant = 'light-to-dark',
   502→}: {
   503→  variant?: 'light-to-dark' | 'dark-to-light';
   504→}) {

  1052→  const [pull, setPull] = useState(0);
  1053→  useEffect(() => {
  1054→    let raf = 0;
  1055→    const onScroll = () => {
  1056→      cancelAnimationFrame(raf);
  1057→      raf = requestAnimationFrame(() => {
  1058→        // Only meaningful at the very top: a negative scrollY (rubber-band)
  1059→        // or a touch overscroll. We approximate with scrollY <= 0.
  1060→        const y = window.scrollY;
  1061→        setPull(y <= 0 ? Math.min(1, (Math.abs(y) + 1) / 60) : 0);
  1062→      });
  1063→    };
  1064→    window.addEventListener('scroll', onScroll, { passive: true });
  1065→    return () => {
  1066→      window.removeEventListener('scroll', onScroll);
  1067→      cancelAnimationFrame(raf);
  1068→    };
  1069→  }, []);
  1070→  if (pull <= 0.02) return null;
  1071→  return (
  1072→    <div
  1073→      aria-hidden
  1074→      className="md:hidden fixed top-1 left-1/2 -translate-x-1/2 z-30 pointer-events-none"
  1075→      style={{ opacity: pull }}
  1076→    >
  1077→      <motion.span
  1078→        animate={{ rotate: 360 }}
  1079→        transition={{ duration: 0.9, repeat: Infinity, ease: 'linear' }}
  1080→        className="block w-6 h-6 rounded-full border-2 border-orange-400/30 border-t-orange-500"
  1081→      />
  1082→    </div>
  1083→  );
  1084→}

  1086→/* ── useSnapActive — tracks which SnapRail card is centered ──
  1087→   Attach the returned ref to a SnapRail's inner scroll container.
  1088→   Returns the active index. */
  1089→export function useSnapActive(count: number) {
  1090→  const ref = useRef<HTMLDivElement>(null);
  1091→  const [active, setActive] = useState(0);
  1092→  const onScroll = useCallback(() => {
  1093→    const el = ref.current;
  1094→    if (!el) return;
  1095→    const center = el.scrollLeft + el.clientWidth / 2;
  1096→    let best = 0;
  1097→    let bestDist = Infinity;
  1098→    Array.from(el.children).forEach((child, i) => {
  1099→      const c = child as HTMLElement;
  1100→      const cCenter = c.offsetLeft + c.clientWidth / 2;
  1101→      const d = Math.abs(cCenter - center);
  1102→      if (d < bestDist) {
  1103→        bestDist = d;
  1104→        best = i;
  1105→      }
  1106→    });
  1107→    setActive(best);
  1108→  }, [count]);
  1109→  useEffect(() => {
  1110→    const el = ref.current;
  1111→    if (!el) return;
  1112→    el.addEventListener('scroll', onScroll, { passive: true });
  1113→    // Defer the initial active-card computation to the next frame so we
  1114→    // don't call setState synchronously inside this effect (cascading render).
  1115→    const raf = requestAnimationFrame(onScroll);
  1116→    return () => {
  1117→      el.removeEventListener('scroll', onScroll);
  1118→      cancelAnimationFrame(raf);
  1119→    };
  1120→  }, [onScroll]);
  1121→  const scrollTo = useCallback(
  1122→    (i: number) => {
  1123→      const el = ref.current;
  1124→      if (!el) return;
  1125→      const child = el.children[i] as HTMLElement | undefined;
  1126→      if (!child) return;
  1127→      el.scrollTo({ left: child.offsetLeft - (el.clientWidth - child.clientWidth) / 2, behavior: 'smooth' });
  1128→    },
  1129→    []
  1130→  );
  1131→  return { ref, active, scrollTo };
  1132→}
