    25→export interface MegaLink {
    26→  label: string;
    27→  href: string;
    28→}
    29→
    30→export interface MegaColumn {
    31→  title: string;
    32→  icon: string;
    33→  /** Flat link list — used when the column has no sub-headings. */
    34→  items: MegaLink[];
    35→  /** Optional sub-headings; when present these replace `items`. */
    36→  groups?: { title: string; items: MegaLink[] }[];
    37→}
    38→
    39→/** Card tile used by the card-style mega menu (see `NavItem.cards`). */
    40→export interface MegaCard {
    41→  icon: string;
    42→  /** Overrides the default navy glyph — e.g. WhatsApp green. */
    43→  iconColor?: string;
    44→  title: string;
    45→  desc: string;
    46→  href: string;
    47→}
    48→
    49→export interface NavItem {
    50→  id: string;
    51→  label: string;
    52→  icon: string;
    53→  href: string;
    54→  columns: MegaColumn[];
    55→  /** When present the panel renders as icon cards instead of link columns. */
    56→  cards?: MegaCard[];
    57→  promo: { title: string; desc: string; cta: string; bg: string; slot?: string };
    58→  features: { icon: string; text: string }[];
    59→}
